cmake_minimum_required(VERSION 3.16)
project(ArmUnicodeConverter LANGUAGES CXX)
if(MSVC)
  add_compile_options(/utf-8)
endif()
add_library(arm_conversion ConvertFunctions.cpp)
target_compile_features(arm_conversion PUBLIC cxx_std_17)
target_include_directories(arm_conversion PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
option(ARM_BUILD_GUI "Build the wxWidgets desktop application" ON)
if(ARM_BUILD_GUI)
  if(WIN32)
    find_package(wxWidgets CONFIG REQUIRED)
    set(ARM_WX_LIBRARIES wx::core wx::base wx::stc)
  else()
    find_package(wxWidgets REQUIRED COMPONENTS core base stc)
    include(${wxWidgets_USE_FILE})
    set(ARM_WX_LIBRARIES ${wxWidgets_LIBRARIES})
  endif()
  add_executable(arm_unicode_converter WIN32 Main.cpp)
  target_link_libraries(arm_unicode_converter PRIVATE arm_conversion ${ARM_WX_LIBRARIES})
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(conversion_tests tests/conversion_tests.cpp)
  target_link_libraries(conversion_tests PRIVATE arm_conversion)
  add_test(NAME conversion COMMAND conversion_tests)
  if(ARM_BUILD_GUI)
    add_executable(gui_tests tests/gui_tests.cpp)
    target_link_libraries(gui_tests PRIVATE arm_conversion ${ARM_WX_LIBRARIES})
    # Run under a desktop display or xvfb-run; excluded from headless CTest.
  endif()
endif()
